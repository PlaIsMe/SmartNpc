cmake_minimum_required(VERSION 3.24)
project(smartnpc_llama LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON ON CACHE BOOL "" FORCE)
set(LLAMA_BUILD_IS_DEV OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMIT "${LLAMA_VERSION}" CACHE STRING "Upstream release tag" FORCE)
set(LLAMA_BUILD_NUMBER 0 CACHE STRING "Release archive build" FORCE)
foreach(feature TESTS TOOLS EXAMPLES SERVER APP MTMD)
    set(LLAMA_BUILD_${feature} OFF CACHE BOOL "" FORCE)
endforeach()
set(LLAMA_OPENSSL OFF CACHE BOOL "" FORCE)
set(LLAMA_SUBPROCESS OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_CCACHE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
set(GGML_AVX2 ON CACHE BOOL "" FORCE)
set(GGML_AVX512 OFF CACHE BOOL "" FORCE)
if(NOT LLAMA_SOURCE OR NOT LLAMA_VERSION OR NOT JAVA_HOME)
    message(FATAL_ERROR "Set LLAMA_SOURCE, LLAMA_VERSION and JAVA_HOME")
endif()
add_subdirectory("${LLAMA_SOURCE}" llama)
add_library(smartnpc_llama SHARED smartnpc_llama.cpp)
target_include_directories(smartnpc_llama PRIVATE "${JAVA_HOME}/include")
if(WIN32)
    target_include_directories(smartnpc_llama PRIVATE "${JAVA_HOME}/include/win32")
elseif(APPLE)
    target_include_directories(smartnpc_llama PRIVATE "${JAVA_HOME}/include/darwin")
else()
    target_include_directories(smartnpc_llama PRIVATE "${JAVA_HOME}/include/linux")
endif()
target_link_libraries(smartnpc_llama PRIVATE llama-common llama)
if(MINGW)
    target_link_options(smartnpc_llama PRIVATE -static -static-libgcc -static-libstdc++)
endif()
set_target_properties(smartnpc_llama PROPERTIES PREFIX "")
